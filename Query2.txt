PREFIX crm: <http://www.cidoc-crm.org/cidoc-crm/>
PREFIX dct: <http://purl.org/dc/terms/>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

SELECT DISTINCT ?object ?objectLabel ?publication ?publicationTitle
WHERE {
  ?object rdf:type crm:E22_Human-Made_Object .
  ?object rdfs:label ?objectLabel .
  ?publication rdf:type/rdfs:subClassOf* dct:BibliographicResource .
  ?publication dct:title ?publicationTitle . 
  ?publication crm:P70_documents ?object .
}
